add_executable(jvconvert main.cpp shared.cpp mcp.cpp)
target_link_libraries(jvconvert PRIVATE jowe::convert)
if(WIN32)
  target_link_libraries(jvconvert PRIVATE shell32)
endif()
if(MSVC)
  target_compile_options(jvconvert PRIVATE /W4 /permissive- /utf-8 /wd4996)
  target_compile_definitions(jvconvert PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(jvconvert PRIVATE -Wall -Wextra)
endif()
if(EMSCRIPTEN)
  target_link_options(jvconvert PRIVATE -sNODERAWFS=1 -sEXIT_RUNTIME=1)
endif()
install(TARGETS jvconvert RUNTIME DESTINATION bin)
